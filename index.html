كل صفحة صفّ في جدول الصفحات، وكل سطر داخلها صفّ في جدول البلوكات. صفوف قواعد البيانات صفحات عادية تحمل قيم خصائصها.

create table pages (
  id          uuid primary key,
  parent_id   uuid references pages,
  kind        text,   -- page | database | row
  title       text,
  icon        text,
  cover       text,
  props       jsonb,  -- row values
  items       jsonb,  -- whiteboard items
  schema      jsonb,  -- columns + views
  position    int,
  trashed_at  timestamptz
);

create table blocks (
  id        uuid primary key,
  page_id   uuid references pages on delete cascade,
  type      text,     -- p, h1, todo, toggle ...
  content   text,     -- simple HTML
  indent    int,
  checked   boolean,
  color     text,
  position  int
);
